public class Primes {
    public static void main(String[] args) {
        int count =0;
        // 逐个判断 0~1000 中的每个数是否为质数
        for (int n = 2; n <= 1000; n++) {
            if (isPrime(n)) {
                System.out.print(n + " ");
                count++
            }
        }
        System.out.println("数量"+count);
    }

    // 判断 n 是否为质数
    static boolean isPrime(int n) {
        if (n < 2) return false;          // 0、1 不是质数
        // 只需检查到 sqrt(n) 即可，减少循环次数
        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) return false; // 能被整除则不是质数
        }
        return true;
    }
}

// 对象
public class Student {
    // ===== 成员变量（属性/字段） =====
    private String name;        // 姓名
    private int age;            // 年龄
    private String major;       // 专业
    private static int count = 0;  // 静态变量：统计创建了多少个对象

    // ===== 构造方法（用于初始化对象） =====
    // 构造器特点：创建对象时，对象会立即自动调用构造器。
    public Student() {
        count++;  // 每创建一个对象，计数加 1
    }
    // this主要用来解决：成员变量与方法内部变量名称冲突问题。
    public Student(String name, int age, String major) {
        this.name = name;
        this.age = age;
        this.major = major;
        count++;
    }

    // ===== 普通方法 =====
    // getter / setter
    public String getName() {
        return name;
    }
    public void setName(String name) {
        this.name = name;
    }

    public int getAge() {
        return age;
    }
    public void setAge(int age) {
        this.age = age;
    }

    public String getMajor() {
        return major;
    }
    public void setMajor(String major) {
        this.major = major;
    }

    // 业务方法：自我介绍
    public void introduce() {
        System.out.println("我叫 " + name + "，今年 " + age + " 岁，专业是 " + major);
    }
    // 静态方法：获取对象总数
    public static int getCount() {
        return count;
    }

    // ===== 覆写 toString()：方便打印对象 =====
    @Override
    public String toString() {
        return "Student{name='" + name + "', age=" + age + ", major='" + major + "'}";
    }
}
public class Main {
    public static void main(String[] args) {
        // 1. 用无参构造创建对象，再通过 setter 赋值
        Student s1 = new Student();
        s1.setName("张三");
        s1.setAge(20);
        s1.setMajor("计算机科学");

        // 2. 用带参构造直接创建并初始化
        Student s2 = new Student("李四", 22, "软件工程");

        // 3. 调用方法
        s1.introduce();                    // 业务方法
        System.out.println(s2.toString()); // 覆写的 toString
        System.out.println("共创建对象数：" + Student.getCount()); // 静态方法
    }
}
